    let projects = JSON.parse(localStorage.getItem('TD_PROJECTS')) || defaultData;
    let isAdminMode = false;

    function saveToStorage() {
      localStorage.setItem('TD_PROJECTS', JSON.stringify(projects));
      updateStats();
      renderProjects();
    }

    function toggleViewMode() {
      isAdminMode = !isAdminMode;
      document.getElementById('admin-toolbar').classList.toggle('hidden', !isAdminMode);
      document.getElementById('view-mode-label').innerText = isAdminMode ? '切換至總經理總覽模式' : '切換至後台管理模式';
      renderProjects();
    }

    function updateStats() {
      const total = projects.length;
      const running = projects.filter(p => p.status === '進行中').length;
      const evalCount = projects.filter(p => p.status === '評估中').length;
      const avgProgress = total > 0 ? (projects.reduce((acc, cur) => acc + (Number(cur.progress) || 0), 0) / total).toFixed(1) : 0;
      
      document.getElementById('stat-total').innerText = total;
      document.getElementById('stat-running').innerText = running;
      document.getElementById('stat-eval').innerText = evalCount;
      document.getElementById('stat-progress').innerText = avgProgress + '%';
    }

    function renderProjects() {
      const q = document.getElementById('search-input').value.toLowerCase();
      const statusF = document.getElementById('filter-status').value;
      const regionF = document.getElementById('filter-region').value;

      const filtered = projects.filter(p => {
        const matchQ = p.customer.toLowerCase().includes(q) || p.owner.toLowerCase().includes(q) || p.item.toLowerCase().includes(q);
        const matchStatus = statusF === 'ALL' || p.status === statusF;
        const matchRegion = regionF === 'ALL' || p.region.includes(regionF);
        return matchQ && matchStatus && matchRegion;
      });

      const container = document.getElementById('project-container');
      container.innerHTML = '';

      filtered.forEach(p => {
        const isRunning = p.status === '進行中';
        const card = document.createElement('div');
        card.className = "bg-slate-800/90 border border-slate-700 rounded-xl p-5 hover:border-slate-500 transition shadow-sm flex flex-col justify-between";
        card.innerHTML = `
          <div>

              <span class="text-xs text-slate-400 font-mono bg-slate-900/60 px-2 py-0.5 rounded">${p.region} · ${p.owner}</span>
            </div>
            <h3 class="font-bold text-base text-white mb-2 leading-snug">${p.customer}</h3>
            <p class="text-xs text-slate-300 mb-4 line-clamp-2">${p.item}</p>
            
            <div class="mb-4">
              <div class="flex justify-between text-xs mb-1">
                <span class="text-slate-400">目前進度</span>
                <span class="text-emerald-400 font-bold">${p.progress}%</span>
              </div>
              <div class="w-full bg-slate-700 h-2 rounded-full overflow-hidden">

      const modal = document.getElementById('edit-modal');
      const form = document.getElementById('project-form');
      form.reset();
      
      if (id) {
        const item = projects.find(p => p.id === id);
        document.getElementById('modal-title').innerText = '編輯專案：' + item.customer;
        document.getElementById('form-id').value = item.id;
        document.getElementById('form-customer').value = item.customer;
        document.getElementById('form-region').value = item.region;
        document.getElementById('form-status').value = item.status;
        document.getElementById('form-owner').value = item.owner;
        document.getElementById('form-progress').value = item.progress;
        document.getElementById('form-item').value = item.item;
        document.getElementById('form-action').value = item.action;
      } else {
        document.getElementById('modal-title').innerText = '新增專案';
        document.getElementById('form-id').value = '';
        document.getElementById('form-progress').value = 0;
      }
      modal.classList.remove('hidden');
    }

    function closeEditModal() {
      document.getElementById('edit-modal').classList.add('hidden');
    }

    function saveProject(e) {
      e.preventDefault();
      const id = document.getElementById('form-id').value;
      const projectData = {
        id: id || 'WT-' + Date.now().toString().slice(-4),
        customer: document.getElementById('form-customer').value,
        region: document.getElementById('form-region').value || '未定',
        status: document.getElementById('form-status').value,
        owner: document.getElementById('form-owner').value || '未指派',
        progress: Number(document.getElementById('form-progress').value) || 0,
        item: document.getElementById('form-item').value,
        action: document.getElementById('form-action').value
      };

      if (id) {
        const idx = projects.findIndex(p => p.id === id);
        if (idx !== -1) projects[idx] = projectData;
      } else {
        projects.unshift(projectData);
      }
      saveToStorage();
      closeEditModal();
    }

    function deleteProject(id) {
      if (confirm('確定要刪除此專案嗎？')) {
        projects = projects.filter(p => p.id !== id);
        saveToStorage();
      }
    }

    function exportJSON() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(projects, null, 2));
      const dl = document.createElement('a');
      dl.setAttribute("href", dataStr);
      dl.setAttribute("download", `溙鼎水務專案資料備份_${new Date().toISOString().slice(0,10)}.json`);
      dl.click();
    }

    function importJSON(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(evt) {
        try {
          const imported = JSON.parse(evt.target.result);
          if (Array.isArray(imported)) {
            projects = imported;
            saveToStorage();
            alert('專案資料已成功匯入！');
          }
        } catch(err) {
          alert('匯入格式有誤，請確認檔案為 JSON。');
        }
      };
      reader.readAsText(file);
    }

    // 初始化載入
    updateStats();
    renderProjects();
